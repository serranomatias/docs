---
title: 'List Calls'
api: 'GET /v1/calls'
description: 'Retrieve a paginated list of calls for your organization'
---

Returns a paginated list of your organization's calls, newest first by default. Each call includes its transcript (`chat_history`), its analysis and its contact.

The only filter is `phoneNumber`, which narrows the list to one contact. This endpoint does not filter by date, status, result, agent or campaign: any query parameter not listed below is rejected with `400`.

<Note>
  This endpoint requires your organization's API key. A personal API key gets `403 Forbidden`.
</Note>

## Headers

<ParamField header="x-api-key" type="string" required>
  Your organization's API key. Found in your dashboard under API settings.
</ParamField>

## Query Parameters

<ParamField query="limit" type="number" default="5">
  Maximum number of calls to return, up to `1000`. If omitted or `0`, returns 5.
</ParamField>

<ParamField query="offset" type="number" default="0">
  Number of calls to skip for pagination. Must be ≥ 0.
</ParamField>

<ParamField query="order" type="string" default="desc">
  Sort direction by creation date. Possible values: `asc`, `desc` (lowercase).
</ParamField>

<ParamField query="phoneNumber" type="string">
  Phone number in E.164 format (e.g. `+12124567890`). Returns only the calls of the contact in your organization that has this phone number. It does not match the number a call was placed from or to.
</ParamField>

## Response

The response contains `statusCode`, a `data` array, and `pagination`.

<ResponseField name="data" type="array">
  Array of call objects.
  <Expandable title="Call object">
    <ResponseField name="id" type="string">
      Unique identifier of the call (UUID).
    </ResponseField>
    <ResponseField name="direction" type="string">
      Call direction (e.g. `outbound-api`). May be `null`.
    </ResponseField>
    <ResponseField name="from" type="string">
      The phone number that placed the call (E.164 format).
    </ResponseField>
    <ResponseField name="chat_history" type="array">
      Transcript of the call, always included. Empty array when there is no transcript.
      <Expandable title="Transcript entry">
        <ResponseField name="role" type="string">Who spoke: `agent` or `user`.</ResponseField>
        <ResponseField name="content" type="string">What was said.</ResponseField>
        <ResponseField name="words" type="array">Per-word timing: `word`, `start` and `end` (seconds from the start of the call).</ResponseField>
        <ResponseField name="metadata" type="object">Present on some agent entries.</ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="call_record" type="string">
      URL to the call recording, if available.
    </ResponseField>
    <ResponseField name="call_status" type="string">
      Current status of the call (e.g. `ended`, `failed`, `in_queue`). May be `null`.
    </ResponseField>
    <ResponseField name="call_result" type="string">
      Outcome of the call (e.g. `answered`, `voicemail`, `no_answer`, `failed`). May be `null`.
    </ResponseField>
    <ResponseField name="disconnection_reason" type="string">
      Reason the call ended (e.g. `user_hangup`, `agent_hangup`, `voicemail`, `max_duration_reached`), if applicable.
    </ResponseField>
    <ResponseField name="start_at" type="string">
      Timestamp when the call started (ISO 8601).
    </ResponseField>
    <ResponseField name="finished_at" type="string">
      Timestamp when the call ended (ISO 8601).
    </ResponseField>
    <ResponseField name="duration" type="string">
      Call duration in seconds, as a string (e.g. `"41"`).
    </ResponseField>
    <ResponseField name="call_analysis" type="object">
      AI-generated analysis of the call. Usually includes `call_summary`, `user_sentiment`, `call_successful` and `custom_analysis_data` (the fields defined in your agent's post-call analysis). Empty when the call was not analyzed.
    </ResponseField>
    <ResponseField name="type" type="string">
      Call type (`api_call`).
    </ResponseField>
    <ResponseField name="created_at" type="string">
      Timestamp when the call record was created (ISO 8601).
    </ResponseField>
    <ResponseField name="contact" type="object">
      Contact associated with the call.
      <Expandable title="Contact object">
        <ResponseField name="full_name" type="string">Contact full name.</ResponseField>
        <ResponseField name="phone_number" type="string">Contact phone number.</ResponseField>
        <ResponseField name="email" type="string">Contact email address.</ResponseField>
        <ResponseField name="country" type="string">Contact country.</ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="pagination" type="object">
  Pagination metadata: `page` (current page, counting from 1, given `limit` and `offset`), `total_rows` and `total_pages`.
</ResponseField>

## Error Codes

- `400 Bad Request` — Invalid query parameters: an unknown parameter (`property <name> should not exist`), `limit` above 1000, an `order` other than `asc`/`desc`, or a `phoneNumber` that is not a valid number. Today it is also returned when `phoneNumber` does not match any contact in your organization.
- `401 Unauthorized` — Invalid or missing API key
- `403 Forbidden` — The key is a personal API key; use your organization's API key
- `429 Too Many Requests` — Rate limit exceeded; honor `Retry-After` when present and use bounded backoff. See [Rate Limits](/api-reference/rate-limits)
- `500 Internal Server Error` — Server-side error

## Code Examples

<RequestExample>
```bash cURL — Basic list
curl -X GET "https://api.contactship.ai/v1/calls?limit=20&offset=0" \
  -H "x-api-key: your-api-key"
```

```bash cURL — One contact, oldest first
curl -G "https://api.contactship.ai/v1/calls" \
  --data-urlencode "phoneNumber=+14155552678" \
  --data-urlencode "order=asc" \
  -H "x-api-key: your-api-key"
```

```javascript JavaScript
const listCalls = async (params = {}) => {
  const query = new URLSearchParams(params).toString();
  const response = await fetch(
    `https://api.contactship.ai/v1/calls${query ? `?${query}` : ''}`,
    { headers: { 'x-api-key': 'your-api-key' } }
  );
  if (!response.ok) throw new Error(`HTTP ${response.status}`);
  return response.json();
};

// Example: the last 10 calls of one contact
const result = await listCalls({
  limit: 10,
  order: 'desc',
  phoneNumber: '+14155552678',
});
console.log(`${result.data.length} calls, ${result.pagination.total_rows} total`);
```

```python Python
import requests

api_key = "your-api-key"

params = {
    "limit": 10,
    "order": "desc",
    "phoneNumber": "+14155552678",
}

response = requests.get(
    "https://api.contactship.ai/v1/calls",
    headers={"x-api-key": api_key},
    params=params,
)

if response.status_code == 200:
    result = response.json()
    print(f"{len(result['data'])} calls returned, {result['pagination']['total_rows']} total")
else:
    print(f"Error {response.status_code}: {response.text}")
```
